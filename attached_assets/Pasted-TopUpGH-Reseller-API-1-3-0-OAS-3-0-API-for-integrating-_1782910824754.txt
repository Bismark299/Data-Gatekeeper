TopUpGH Reseller API
 1.3.0 
OAS 3.0
API for integrating with TopUpGH platform to create data orders programmatically.

Authentication Required:

API Key (X-API-Key): Your API key from My Account > API Management
Timestamp (X-Timestamp): Current Unix timestamp
API Signature (X-API-Signature): HMAC-SHA256 signature (REQUIRED for POST requests)
How to Generate Signature:

signature_string = timestamp + method + endpoint + body
signature = HMAC-SHA256(signature_string, API_SECRET)
Example (POST /orders/create):

const timestamp = Math.floor(Date.now() / 1000);
const method = "POST";
// IMPORTANT: Use the internal route for signature, NOT the clean URL
const endpoint = "/topupgh-api/v1/orders/create";
const body = JSON.stringify({orders: [...]});
const signatureString = timestamp + method + endpoint + body;
const signature = crypto.createHmac("sha256", API_SECRET).update(signatureString).digest("hex");
⚠️ Important:

API Secret is found in My Account > API Management
HMAC signature is REQUIRED for order creation (POST requests)
Use /topupgh-api/v1/... for signature (NOT /api/v1/...)
The Swagger UI auto-generates signatures when you enter your API Secret
Domain Whitelist Required: You must add your domain/server IP in My Account > API Management > Domain Whitelist. Without a whitelist, all API requests will be blocked.

/test - Test API Connection
const crypto = require("crypto");

const API_KEY = "YOUR_API_KEY";
const API_SECRET = "YOUR_API_SECRET";
const BASE_URL = "https://reseller.etopupgh.com/api/v1";

async function testConnection() {
    const timestamp = Math.floor(Date.now() / 1000).toString();
    const method = "GET";
    const endpoint = "/topupgh-api/v1/test";
    const body = "";
    
    const signatureString = timestamp + method + endpoint + body;
    const signature = crypto
        .createHmac("sha256", API_SECRET)
        .update(signatureString)
        .digest("hex");
    
    const response = await fetch(BASE_URL + "/test", {
        method: "GET",
        headers: {
            "Accept": "application/json",
            "X-API-Key": API_KEY,
            "X-Timestamp": timestamp,
            "X-API-Signature": signature
        }
    });
    
    const data = await response.json();
    console.log(data);
}

/orders/create - Create Bulk Order
const crypto = require("crypto");

const API_KEY = "YOUR_API_KEY";
const API_SECRET = "YOUR_API_SECRET";
const BASE_URL = "https://reseller.etopupgh.com/api/v1";

async function createOrder() {
    const orders = [
        { _beneficiary_number: "0241234567", network: "mtn", _data_size: 5 },
        { _beneficiary_number: "0551234567", network: "at", _data_size: 10 }
    ];
    
    const timestamp = Math.floor(Date.now() / 1000).toString();
    const method = "POST";
    const endpoint = "/topupgh-api/v1/orders/create";
    const body = JSON.stringify({ orders });
    
    const signatureString = timestamp + method + endpoint + body;
    const signature = crypto
        .createHmac("sha256", API_SECRET)
        .update(signatureString)
        .digest("hex");
    
    const response = await fetch(BASE_URL + "/orders/create", {
        method: "POST",
        headers: {
            "Accept": "application/json",
            "Content-Type": "application/json",
            "X-API-Key": API_KEY,
            "X-Timestamp": timestamp,
            "X-API-Signature": signature
        },
        body: body
    });
    
    const data = await response.json();
    console.log("Order ID:", data.order_id);
}

// Express.js webhook endpoint example
const express = require("express");
const app = express();

app.use(express.json());

app.post("/webhook", (req, res) => {
    const payload = req.body;
    
    // Sample payload structure:
    // {
    //     "event": "delivery_status_updated",
    //     "timestamp": "2026-01-17T10:30:00+00:00",
    //     "order": {
    //         "order_id": 123456,
    //         "order_number": "123456",
    //         "delivery_info": "Submited to the Yello server on 7:24AM - 14/January/2026",
    //         "delivery_date": "2026-01-14",
    //         "delivery_time": "07:24:00",
    //         "formatted_delivery_info": "Submited to the Yello server on 7:24AM - 14/January/2026",
    //         "items": [
    //             {
    //                 "item_id": "123456-1",
    //                 "beneficiary_number": "0241234567",
    //                 "network": "mtn",
    //                 "data_size": 5,
    //                 "delivery_status": "Delivered",
    //                 "delivery_date": "2026-01-14",
    //                 "delivery_time": "07:24:00"
    //             }
    //         ]
    //     }
    // }
    
    if (payload.event === "delivery_status_updated") {
        const orderId = payload.order.order_id;
        const deliveryInfo = payload.order.delivery_info;
        // Process delivery status update
        console.log("Order", orderId, "delivered:", deliveryInfo);
    }
    
    res.status(200).send("OK");
});

app.listen(3000, () => {
    console.log("Webhook server listening on port 3000");
});
